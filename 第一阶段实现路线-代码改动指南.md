"""Power cycle test state and configuration."""
from dataclasses import dataclass, field
from enum import Enum
from datetime import datetime


class CycleState(Enum):
    """循环执行状态."""
    IDLE = "就绪"
    RUNNING = "运行中"
    PAUSED = "已暂停"
    STOPPING = "停止中"
    COMPLETED = "已完成"


@dataclass
class PowerCycleConfig:
    """断电循环配置参数."""
    total_cycles: int = 10
    interval_seconds: int = 15  # 两次断电的间隔
    enable_pre_warning: bool = True  # 断电前5秒警告
    stop_on_error: bool = False  # 错误时是否停止
    
    def validate(self) -> tuple[bool, str]:
        """验证参数有效性."""
        if self.total_cycles < 1:
            return False, "循环次数至少为 1"
        if self.interval_seconds < 5:
            return False, "安全间隔至少 5 秒"
        return True, ""


@dataclass
class PowerCycleProgress:
    """循环执行进度与统计."""
    current_cycle: int = 0
    completed_cycles: int = 0
    failed_cycles: int = 0
    skipped_cycles: int = 0
    started_at: datetime | None = None
    paused_at: datetime | None = None
    resumed_at: datetime | None = None
    completed_at: datetime | None = None
    
    @property
    def elapsed_seconds(self) -> float:
        """已耗时（秒）."""
        if self.started_at is None:
            return 0.0
        end = self.completed_at or self.paused_at or datetime.now()
        return (end - self.started_at).total_seconds()
    
    @property
    def success_rate(self) -> float:
        """成功率 (0-100)."""
        total = self.completed_cycles + self.failed_cycles
        if total == 0:
            return 0.0
        return (self.completed_cycles / total) * 100


@dataclass
class CycleEvent:
    """循环执行事件记录."""
    cycle_number: int
    event_type: str  # "power_off", "power_on", "check", "error", "skip"
    timestamp: datetime
    detail: str = ""
    duration_ms: int = 0
```

from PyQt6.QtWidgets import QMessageBox, QWidget
from config import ROBOT_SCENARIOS, ACTION_RISK_LEVEL


class ControlPanel(QWidget):
    """基础控制面板，增加吊装保护."""
    
    def __init__(self, robot_service=None, parent=None):
        super().__init__(parent)
        self._service = robot_service
        self._current_scenario = "ground"  # 当前工况
        self._is_hoisting = False
        self._build_ui()
    
    def set_scenario(self, scenario_key: str):
        """切换工况."""
        if scenario_key not in ROBOT_SCENARIOS:
            return
        self._current_scenario = scenario_key
        self._is_hoisting = (scenario_key == "hoisting")
    
    def execute_action(self, action_name: str) -> bool:
        """
        执行动作前检查工况合法性.
        
        Returns:
            True 表示动作允许执行，False 表示被拦截
        """
        scenario = ROBOT_SCENARIOS.get(self._current_scenario)
        if not scenario:
            return False
        
        # 检查是否在禁止列表
        if action_name in scenario["prohibited_actions"]:
            risk_level = ACTION_RISK_LEVEL.get(action_name, "medium")
            
            if risk_level == "critical":
                # 最高风险，直接禁止
                QMessageBox.critical(
                    self, "动作被拦截",
                    f"动作 '{action_name}' 在 '{scenario['name']}' 工况下风险极高，已被禁止。"
                )
                return False
            
            elif risk_level == "high":
                # 高风险，弹窗确认
                reply = QMessageBox.warning(
                    self, "动作风险警告",
                    f"动作 '{action_name}' 在 '{scenario['name']}' 工况下风险较高。\n\n"
                    f"确认继续执行吗？",
                    QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
                    QMessageBox.StandardButton.No
                )
                if reply != QMessageBox.StandardButton.Yes:
                    return False
        
        # 前置检查通过，调用服务执行
        try:
            result = self._service.execute_action(action_name)
            if not result.success:
                self.log_message.emit(
                    "error",
                    f"动作 '{action_name}' 执行失败: {result.error}"
                )
            return result.success
        except Exception as e:
            QMessageBox.critical(self, "执行错误", str(e))
            return False

"""验收会话与结果记录."""
from dataclasses import dataclass, asdict, field
from datetime import datetime
from enum import Enum
import json
from pathlib import Path


class CheckStatus(Enum):
    """检查结果状态."""
    PASS = "pass"
    FAIL = "fail"
    SKIP = "skip"
    TIMEOUT = "timeout"


@dataclass
class AcceptanceCheckResult:
    """单项检查结果."""
    check_key: str  # e.g., "wifi", "main_ssh"
    category: str  # e.g., "网络", "SSH"
    name: str  # 检查名称
    status: CheckStatus
    detail: str = ""  # 返回的详细信息
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())
    duration_ms: int = 0


@dataclass
class AcceptanceSession:
    """验收会话记录."""
    session_id: str  # YYYYMMDD_HHMMSS_<robot_model>
    robot_model: str = "unknown"
    robot_sn: str = ""  # 序列号
    started_at: str = field(default_factory=lambda: datetime.now().isoformat())
    completed_at: str | None = None
    results: list[AcceptanceCheckResult] = field(default_factory=list)
    notes: str = ""  # 用户备注
    
    @staticmethod
    def create(robot_model: str = "unknown") -> "AcceptanceSession":
        """创建新会话."""
        session_id = datetime.now().strftime("%Y%m%d_%H%M%S_") + robot_model
        return AcceptanceSession(session_id=session_id, robot_model=robot_model)
    
    def add_result(self, result: AcceptanceCheckResult):
        """添加检查结果."""
        self.results.append(result)
    
    def complete(self):
        """标记会话完成."""
        self.completed_at = datetime.now().isoformat()
    
    @property
    def pass_count(self) -> int:
        """通过的检查数."""
        return sum(1 for r in self.results if r.status == CheckStatus.PASS)
    
    @property
    def fail_count(self) -> int:
        """失败的检查数."""
        return sum(1 for r in self.results if r.status == CheckStatus.FAIL)
    
    @property
    def skip_count(self) -> int:
        """跳过的检查数."""
        return sum(1 for r in self.results if r.status == CheckStatus.SKIP)
    
    @property
    def total_count(self) -> int:
        """总检查数."""
        return len(self.results)
    
    @property
    def pass_rate(self) -> float:
        """通过率 (0-100)."""
        if self.total_count == 0:
            return 0.0
        return (self.pass_count / (self.total_count - self.skip_count)) * 100 if (self.total_count - self.skip_count) > 0 else 0.0
    
    def to_dict(self) -> dict:
        """转换为字典."""
        return {
            "session_id": self.session_id,
            "robot_model": self.robot_model,
            "robot_sn": self.robot_sn,
            "started_at": self.started_at,
            "completed_at": self.completed_at,
            "notes": self.notes,
            "summary": {
                "total": self.total_count,
                "pass": self.pass_count,
                "fail": self.fail_count,
                "skip": self.skip_count,
                "pass_rate": f"{self.pass_rate:.1f}%",
            },
            "results": [asdict(r) for r in self.results],
        }
    
    def save_to_json(self, save_dir: str = None) -> Path:
        """保存为 JSON 文件."""
        if save_dir is None:
            save_dir = str(Path.home() / ".OliRobotManager" / "acceptance_results")
        
        save_path = Path(save_dir) / f"{self.session_id}.json"
        save_path.parent.mkdir(parents=True, exist_ok=True)
        
        with open(save_path, "w", encoding="utf-8") as f:
            json.dump(self.to_dict(), f, ensure_ascii=False, indent=2)
        
        return save_path
    
    def export_to_csv(self, save_dir: str = None) -> Path:
        """导出为 CSV（与飞书表格兼容）."""
        if save_dir is None:
            save_dir = str(Path.home() / ".OliRobotManager" / "acceptance_results")
        
        import csv
        save_path = Path(save_dir) / f"{self.session_id}.csv"
        save_path.parent.mkdir(parents=True, exist_ok=True)
        
        with open(save_path, "w", newline="", encoding="utf-8-sig") as f:
            writer = csv.writer(f)
            writer.writerow(["验收会话", self.session_id])
            writer.writerow(["机器人型号", self.robot_model])
            writer.writerow(["序列号", self.robot_sn])
            writer.writerow(["开始时间", self.started_at])
            writer.writerow(["完成时间", self.completed_at or "进行中"])
            writer.writerow(["通过率", f"{self.pass_rate:.1f}%"])
            writer.writerow([])
            
            writer.writerow(["测试类别", "测试项目", "工具/目标", "结果", "详情", "耗时(ms)"])
            for result in self.results:
                writer.writerow([
                    result.category,
                    result.name,
                    result.check_key,
                    result.status.value,
                    result.detail[:100],  # 截断避免超长
                    result.duration_ms,
                ])
        
        return save_path
```
